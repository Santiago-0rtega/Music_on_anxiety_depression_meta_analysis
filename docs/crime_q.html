CRIME-Q study-level rating figure

The source assessment contains 49 cohort rows from 20 studies. When a study has multiple cohorts, item ratings are collapsed to one display row: if all cohorts share the same rating, that rating is kept; if cohort ratings differ, the study is shown as Partly.

plot_dat <- study_matrix |>
  pivot_longer(
    cols = all_of(item_cols),
    names_to = "Item",
    values_to = "Rating"
  ) |>
  mutate(
    Item = factor(Item, levels = item_cols),
    Study = factor(Study, levels = rev(unique(Study))),
    Fill_group = case_when(
      Rating == "Yes" ~ "Yes",
      Rating %in% c("Partly", "Unclear") ~ "Partly/Unclear",
      Rating == "No" ~ "No",
      Rating == "NA" ~ "NA",
      TRUE ~ "Other"
    )
  )

if (any(plot_dat$Fill_group == "Other")) {
  warning("Some unexpected CRIME-Q rating values were found.")
}


rating_colors <- c(
  "Yes" = "#6F8542",
  "Partly/Unclear" = "#B8842F",
  "No" = "#A64747",
  "NA" = "#557993"
)
gallery_text <- "#242321"

CRIME<-ggplot(plot_dat, aes(x = Item, y = Study, fill = Fill_group)) +
  geom_tile(color = "#FFFDF8", linewidth = 0.6) +
  scale_fill_manual(
    values = rating_colors,
    breaks = c("Yes", "Partly/Unclear", "No", "NA"),
    name = NULL
  ) +
  labs(
    x = NULL,
    y = NULL
  ) +
  theme_minimal(base_size = 12) +
  theme(
    plot.background = element_rect(fill = "transparent", color = NA),
    panel.background = element_rect(fill = "transparent", color = NA),
    legend.background = element_rect(fill = "transparent", color = NA),
    legend.key = element_rect(fill = "transparent", color = NA),
    panel.grid = element_blank(),
    axis.text.x = element_text(
      angle = 0,

    ),

```{r}
#| label: crimeq-load
#| include: false

input_file <- here("..", "data", "MUSIC-CRIME-Q_RoB_assessment.xlsx")
input_sheet <- "Table 2"
codebook_file <- here("..", "CRIME-Q_Music_RoB_Codebook_v1.8.csv")

item_cols <- c(
  "1X", "2X", "2Y", "3X", "3Y", "3Z", "4Y",
  "5X", "5Y", "5Z(1)", "5Z(2)", "5Z(3)",
  "6X", "7X", "7Z(1)", "7Z(2)", "7Z(3)",
  "8X", "8Z(1)", "8Z(2)", "9X", "10X", "10Z"
)

expected_cols <- c("Study_ID", "Cohort_ID", item_cols)

cohort_matrix <- read_excel(
  path = input_file,
  sheet = input_sheet,
  col_types = "text",
  na = ""
) |>
  rename_with(str_trim) |>
  mutate(across(everything(), str_trim)) |>
  select(all_of(expected_cols))

if (!identical(names(cohort_matrix), expected_cols)) {
  stop("Sheet 'Table 2' does not match the expected column structure.")
}

if (anyDuplicated(cohort_matrix$Cohort_ID) > 0) {
  stop("Duplicated Cohort_ID values detected.")
}

collapse_rating <- function(x) {
  vals <- unique(x)
  if (length(vals) == 1L) vals else "Partly"
}

# Turn a Study_ID like "Camargo_2013_PSYN" into "Camargo et al. 2013".
study_label <- function(study_id) {
  parts <- str_split(study_id, "_", simplify = FALSE)
  vapply(parts, function(p) {
    author <- p[1]
    year <- p[2]
    paste0(author, " et al. ", year)
  }, character(1))
}

study_matrix <- cohort_matrix |>
  group_by(Study_ID) |>
  summarise(
    across(all_of(item_cols), collapse_rating),
    .groups = "drop"
  ) |>
  arrange(Study_ID) |>
  mutate(Study = study_label(Study_ID)) |>
  relocate(Study, .before = Study_ID)

codebook <- read_csv(
  codebook_file,
  show_col_types = FALSE,
  trim_ws = TRUE
) |>
  arrange(Order)

if (!identical(codebook$Item, item_cols)) {
  stop("Codebook item order does not match the CRIME-Q matrix columns.")
}
```

## CRIME-Q study-level rating figure

The source assessment contains `r nrow(cohort_matrix)` cohort rows from `r n_distinct(cohort_matrix$Study_ID)` studies. When a study has multiple cohorts, item ratings are collapsed to one display row: if all cohorts share the same rating, that rating is kept; if cohort ratings differ, the study is shown as `Partly`.
